5. Linux 最重要的概念之一：文件描述符 fd#

Linux 中很多资源都可以用一个整数表示：

文件
Socket
管道
设备
    ↓
文件描述符 fd

例如：

int fd = 3;

可以理解为：

3 → 某个已经打开的资源

程序启动后默认有：

0 → stdin
1 → stdout
2 → stderr

也就是：

标准输入
标准输出
标准错误

8. 文件写入实验#

#include <stdio.h>
#include <string.h>
#include <fcntl.h>
#include <unistd.h>

int main()
{
    int fd = open(
        "test.txt",
        O_WRONLY | O_CREAT | O_TRUNC,
        0644
    );

    if(fd < 0)
    {
        perror("open");
        return 1;
    }

    const char *msg = "Hello Linux System Call!\n";

    ssize_t n = write(fd, msg, strlen(msg));

    if(n < 0)
    {
        perror("write");
        close(fd);
        return 1;
    }

    close(fd);
    return 0;
}

编译运行：

9. 文件读取实验#

#include <stdio.h>
#include <fcntl.h>
#include <unistd.h>

int main()
{
    int fd = open("test.txt", O_RDONLY);

    if(fd < 0)
    {
        perror("open");
        return 1;
    }

    char buf[128];

    ssize_t n = read(fd, buf, sizeof(buf) - 1);

    if(n < 0)
    {

}

11. 进程#

程序和进程不同：

程序 = 磁盘上的文件

每个进程都有：

PID

获取 PID：

}

12. fork 创建子进程#

#include <stdio.h>
#include <unistd.h>

int main()
{
    pid_t pid = fork();

    if(pid < 0)
    {
        perror("fork");
        return 1;
    }

    if(pid == 0)
    {
        printf("child, PID=%d\n", getpid());
    }
    else
    {
        printf(
            "parent, PID=%d, child=%d\n",
            getpid(),
            pid
        );
    }

    return 0;
}

记住：

pid < 0  创建失败
pid = 0  当前是子进程
pid > 0  当前是父进程

15. pthread 创建线程#

#include <stdio.h>
#include <pthread.h>

void *worker(void *arg)
{
    (void)arg;
    printf("worker running\n");
    return NULL;
}

int main()
{
    pthread_t tid;

    pthread_create(
        &tid,
        NULL,
        worker,
        NULL
    );

    pthread_join(tid, NULL);

    return 0;
}

编译：

gcc thread_demo.c -o thread_demo -pthread

16. Race Condition#

假设：

int count = 0;

两个线程都执行：

count++;

count++ 可以粗略理解成：

读取 count
加 1
写回 count

可能出现：

A 读取 0
B 读取 0
A 写入 1
B 写入 1

最后结果：

1

而不是 2。

这叫：

Race Condition

17. Mutex 互斥锁#

定义：

pthread_mutex_t mutex =
    PTHREAD_MUTEX_INITIALIZER;

加锁：

pthread_mutex_lock(&mutex);

解锁：

pthread_mutex_unlock(&mutex);

例如：

pthread_mutex_lock(&mutex);

count++;

pthread_mutex_unlock(&mutex);

这段被保护的代码叫：

临界区

18. Mutex 完整例子#

#include <stdio.h>
#include <pthread.h>

long long count = 0;

pthread_mutex_t mutex =
    PTHREAD_MUTEX_INITIALIZER;

void *worker(void *arg)
{
    (void)arg;

    for(int i = 0; i < 1000000; i++)
    {
        pthread_mutex_lock(&mutex);

        count++;

        pthread_mutex_unlock(&mutex);
    }

    return NULL;
}

int main()
{
    pthread_t t1, t2;

    pthread_create(&t1, NULL, worker, NULL);
    pthread_create(&t2, NULL, worker, NULL);

    pthread_join(t1, NULL);
    pthread_join(t2, NULL);

    printf("count = %lld\n", count);

    pthread_mutex_destroy(&mutex);

    return 0;
}

19. 死锁#

经典情况：

线程 A：
拿 lock1
等 lock2

线程 B：
拿 lock2
等 lock1

双方永远等待。

避免死锁最常见方法：

1. 固定加锁顺序
2. 减少同时持有多个锁
3. 临界区尽量短
4. 必要时使用 trylock

最重要：

所有线程统一锁顺序

20. 条件变量#

错误的等待方式：

while(!ready)
{
}

这会一直占 CPU。

条件变量允许线程：

没数据时睡眠
有数据后被唤醒

定义：

pthread_cond_t cond =
    PTHREAD_COND_INITIALIZER;

等待：

pthread_cond_wait(&cond, &mutex);

唤醒：

pthread_cond_signal(&cond);

21. 条件变量标准模板#

消费者：

pthread_mutex_lock(&mutex);

while(!ready)
{
    pthread_cond_wait(
        &cond,
        &mutex
    );
}

处理数据;

pthread_mutex_unlock(&mutex);

生产者：

pthread_mutex_lock(&mutex);

修改共享数据;
ready = 1;

pthread_cond_signal(&cond);

pthread_mutex_unlock(&mutex);

注意：

通常使用 while，而不是 if

因为线程被唤醒后要重新检查条件。

23. Mutex 与 Semaphore#

Mutex
→ 强调互斥
→ 通常一次一个线程

24. 共享内存#

进程默认不能直接共享普通变量。

如果需要高速交换大量数据，可以使用：

Shared Memory

结构：

进程 A
    \
     Shared Memory
    /
进程 B

常见 API：

shm_open()
ftruncate()
mmap()
munmap()
shm_unlink()

流程：

shm_open
↓
ftruncate
↓
mmap
↓
读写内存
↓
munmap

共享内存只解决：

能否看到同一份数据

它不自动解决：

谁先读
谁先写
同时修改怎么办

所以常常还要配合同步机制。

29. TCP Server 示例#

#include <stdio.h>
#include <string.h>
#include <unistd.h>
#include <arpa/inet.h>
#include <sys/socket.h>

int main()
{
    int server_fd = socket(
        AF_INET,
        SOCK_STREAM,
        0
    );

    if(server_fd < 0)
    {
        perror("socket");
        return 1;
    }

    struct sockaddr_in addr = {0};

    addr.sin_family = AF_INET;
    addr.sin_addr.s_addr = htonl(INADDR_ANY);
    addr.sin_port = htons(8888);

    if(bind(
        server_fd,
        (struct sockaddr *)&addr,
        sizeof(addr)
    ) < 0)
    {
        perror("bind");
        return 1;
    }

    if(listen(server_fd, 5) < 0)
    {
        perror("listen");
        return 1;
    }

    printf("listening on 8888...\n");

    int client_fd = accept(
        server_fd,
        NULL,
        NULL
    );

    if(client_fd < 0)
    {
        perror("accept");
        return 1;
    }

    char buf[1024];

    ssize_t n = read(
        client_fd,
        buf,
        sizeof(buf) - 1
    );

    if(n > 0)
    {
        buf[n] = '\0';
        printf("client: %s\n", buf);
    }

    const char *reply = "hello client";

    write(
        client_fd,
        reply,
        strlen(reply)
    );

    close(client_fd);
    close(server_fd);

    return 0;
}

30. TCP Client 示例#

#include <stdio.h>
#include <string.h>
#include <unistd.h>
#include <arpa/inet.h>
#include <sys/socket.h>

int main()
{
    int fd = socket(
        AF_INET,
        SOCK_STREAM,
        0
    );

    struct sockaddr_in server = {0};

    server.sin_family = AF_INET;
    server.sin_port = htons(8888);

    inet_pton(
        AF_INET,
        "127.0.0.1",
        &server.sin_addr
    );

    if(connect(
        fd,
        (struct sockaddr *)&server,
        sizeof(server)
    ) < 0)
    {
        perror("connect");
        return 1;
    }

    const char *msg = "hello server";

    write(fd, msg, strlen(msg));

    char buf[1024];

    ssize_t n = read(
        fd,
        buf,
        sizeof(buf) - 1
    );

    if(n > 0)
    {
        buf[n] = '\0';
        printf("server: %s\n", buf);
    }

    close(fd);

    return 0;
}

34. select#

核心思想：

fd1
fd2
fd3
fd4
 ↓
select
 ↓
哪些 fd 有数据？

常见：

FD_ZERO(&readfds);
FD_SET(fd, &readfds);
FD_ISSET(fd, &readfds);

基本模板：

fd_set readfds;

while(1)
{
    FD_ZERO(&readfds);

    FD_SET(fd1, &readfds);
    FD_SET(fd2, &readfds);

    int maxfd = fd1 > fd2 ? fd1 : fd2;

    int ret = select(
        maxfd + 1,
        &readfds,
        NULL,
        NULL,
        NULL
    );

    if(ret < 0)
    {
        perror("select");
        break;
    }

    if(FD_ISSET(fd1, &readfds))
    {
        // fd1 可读
    }

    if(FD_ISSET(fd2, &readfds))
    {
        // fd2 可读
    }
}

35. select 的缺点#

先掌握三点：

1. fd 数量通常有限制
2. 每轮都要重新准备 fd 集合
3. 返回后通常要遍历检查多个 fd

36. epoll#

Linux 为大量 fd 提供：

epoll

三个核心 API：

epoll_create1()
epoll_ctl()
epoll_wait()

思路：

第一次：
告诉内核我要关注哪些 fd

以后：
epoll_wait 等待

内核：
直接返回发生事件的 fd

37. epoll 基本模板#

#include <sys/epoll.h>

int epfd = epoll_create1(0);

struct epoll_event ev;

ev.events = EPOLLIN;
ev.data.fd = listen_fd;

epoll_ctl(
    epfd,
    EPOLL_CTL_ADD,
    listen_fd,
    &ev
);

struct epoll_event events[64];

while(1)
{
    int n = epoll_wait(
        epfd,
        events,
        64,
        -1
    );

    for(int i = 0; i < n; i++)
    {
        int fd = events[i].data.fd;

        if(fd == listen_fd)
        {
            // accept 新客户端
        }
        else
        {
            // read 客户端数据
        }
    }
}

38. select 与 epoll 区别#

项目 select epoll
学习难度 较低 稍高
平台 多平台 Linux
fd 数量 通常有限制 更适合大量连接
每轮重新传 fd 集合 是 通常不需要
是否常遍历大量 fd 是 主要处理就绪事件
高并发 一般 更适合

可以这样理解：

select：
“这 1000 个人里谁举手了？”
然后自己一个个检查。

epoll：
先登记这 1000 个人，
系统直接告诉你：
“3号、27号、811号举手了。”

39. LT 与 ET#

epoll 常见：

LT
ET

LT：

Level Triggered

默认模式。

只要 fd 仍然可读，就会继续提醒。

ET：

Edge Triggered

通常需要配合：

非阻塞 IO

并一直读取到：

EAGAIN

初学阶段：

先掌握 LT
再学 ET
